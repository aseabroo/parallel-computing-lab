cmake_minimum_required(VERSION 3.16)
project(parallel_computing_lab LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
find_package(OpenMP)
add_executable(cpu_lab src/cpu_lab.cpp)
target_compile_options(cpu_lab PRIVATE -Wall -Wextra -Wpedantic)
if(OpenMP_CXX_FOUND)
  target_link_libraries(cpu_lab PRIVATE OpenMP::OpenMP_CXX)
endif()
if(BUILD_TESTING)
  add_test(NAME cpu_selftest COMMAND cpu_lab --self-test)
endif()
find_package(MPI QUIET COMPONENTS CXX)
if(MPI_CXX_FOUND)
  add_executable(mpi_signal src/mpi_signal.cpp)
  target_link_libraries(mpi_signal PRIVATE MPI::MPI_CXX)
  if(BUILD_TESTING AND MPIEXEC_EXECUTABLE)
    add_test(NAME mpi_four_ranks COMMAND ${MPIEXEC_EXECUTABLE} ${MPIEXEC_NUMPROC_FLAG} 4 ${MPIEXEC_PREFLAGS} $<TARGET_FILE:mpi_signal> 17 ${MPIEXEC_POSTFLAGS})
    add_test(NAME mpi_more_ranks_than_samples COMMAND ${MPIEXEC_EXECUTABLE} ${MPIEXEC_NUMPROC_FLAG} 4 ${MPIEXEC_PREFLAGS} $<TARGET_FILE:mpi_signal> 2 ${MPIEXEC_POSTFLAGS})
  endif()
endif()
